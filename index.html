    function local(a, p, r) {

      let map = { Tarea: 'tareas', Peso: 'pesos', Entrenamiento: 'entrenamientos', Habito: 'habitos', Diario: 'diario', Meta: 'metas' };

      let m = a.match(/(Tarea|Peso|Entrenamiento|Habito|Diario|Meta)$/), list = m ? map[m[1]] : null;

      if (a.startsWith('add') && list) DB[list].push(r);

      if (a.startsWith('update') && list) {

        let i = DB[list].findIndex(x => x.id === p.id);

        if (i > -1) DB[list][i] = { ...DB[list][i], ...r };

      }

      if (a.startsWith('delete') && list) {

        DB[list] = DB[list].filter(x => x.id !== p.id);

        if (a === 'deleteHabito') DB.habitoLogs = DB.habitoLogs.filter(x => String(x.habito_id) !== String(p.id));

      }

      if (a === 'toggleTarea') {

        let x = DB.tareas.find(x => x.id === p.id);

        if (x) x.hecha = p.hecha;

      }

      if (a === 'toggleMeta') {

        let x = DB.metas.find(x => x.id === p.id);

        if (x) x.concluida = p.concluida;

      }

      if (a === 'toggleHabitoLog') {

        let i = DB.habitoLogs.findIndex(x => String(x.habito_id) === String(p.habito_id) && x.fecha === p.fecha);

        if (i > -1) DB.habitoLogs.splice(i, 1);

        else DB.habitoLogs.push({ id: 'temp', habito_id: p.habito_id, fecha: p.fecha });

      }

      if (a === 'saveDesafioConfig') {

        DB.desafioConfig = Object.assign({}, DB.desafioConfig, r);

      }

      if (a === 'toggleDesafioMes') {

        let i = DB.desafioMeses.findIndex(x => Number(x.mes_num) === Number(p.mes_num));

        if (i > -1) DB.desafioMeses[i] = Object.assign({}, DB.desafioMeses[i], r);

        else DB.desafioMeses.push(r);

      }

    }

    // Cálculo da progressão do Desafio (N meses)

    function calcDesafio() {

      const cfg = DB.desafioConfig || {};

      const startStr = (cfg.data_inicio && /^\d{4}-\d{2}-\d{2}$/.test(cfg.data_inicio)) ? cfg.data_inicio : today();

      const totalMeses = Math.max(1, Number(cfg.total_meses) || 24);

      const vIni = Number(cfg.valor_inicial) || 0;

      const inc = Number(cfg.incremento_mensal) || 0;

 

      const [sy, sm] = startStr.split('-').map(Number);

      const nowYm = today().slice(0, 7);

 

      let meses = [];

      let totalProjetado = 0;

      let totalAcumulado = 0;

      let cumpridosCount = 0;

 

      for (let i = 1; i <= totalMeses; i++) {

        const metaValor = vIni + (i - 1) * inc;

        totalProjetado += metaValor;

 

        const d = new Date(sy, (sm - 1) + (i - 1), 1);

        const ym = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;

        const labelMes = d.toLocaleDateString('pt-BR', { month: 'short', year: '2-digit' }).replace('.', '');

 

        const reg = DB.desafioMeses.find(x => Number(x.mes_num) === i);

        const cumprido = reg ? (reg.cumprido === true || reg.cumprido === 'true') : false;

 

        if (cumprido) {

          cumpridosCount++;

          totalAcumulado += (Number(reg.valor_realizado) > 0 ? Number(reg.valor_realizado) : metaValor);

        }

 

        meses.push({

          num: i,

          ym,

          labelMes,

          metaValor,

          cumprido,

          isCurrent: ym === nowYm

        });

      }

 

      // Soma também metas financeiras extras concluídas ou planejadas

      const metasFin = DB.metas.filter(m => m.categoria === 'financeira');

      const metasFinAlvo = metasFin.reduce((s, m) => s + (Number(m.valor_alvo) || 0), 0);

      const metasFinAtual = metasFin.reduce((s, m) => s + (m.concluida ? (Number(m.valor_alvo) || Number(m.valor_atual) || 0) : (Number(m.valor_atual) || 0)), 0);

 

      return {

        startStr,

        totalMeses,

        vIni,

        inc,

        meses,

        cumpridosCount,

        totalProjetado,

        totalAcumulado,

        metasFinAlvo,

        metasFinAtual,

        pct: totalMeses ? Math.round((cumpridosCount / totalMeses) * 100) : 0

      };

    }

    function dashboard(title, value, label, pct = 0) {

      return `<div class="module-head"><h1>${title}</h1><p>Seu progresso, no seu ritmo.</p></div><div class="dash"><span class="eyebrow">${label}</span><div class="metric">${value}</div>${pct ? `<div class="progress"><i style="--p:${pct}%"></i></div>` : ''}</div>`;

    function weightChart() {

      let pts = DB.pesos.slice().sort((a, b) => a.fecha.localeCompare(b.fecha)).slice(-12);

      if (pts.length < 2) return '<div class="empty">Adicione pelo menos 2 registros de peso para ver a evolução.</div>';

      let w = 280, h = 84, pad = 8, vals = pts.map(p => +p.kg), min = Math.min(...vals), max = Math.max(...vals), range = (max - min) || 1, stepX = pts.length > 1 ? (w - pad * 2) / (pts.length - 1) : 0, coords = vals.map((v, i) => [pad + i * stepX, pad + (1 - (v - min) / range) * (h - pad * 2)]), path = coords.map((c, i) => (i ? 'L' : 'M') + c[0].toFixed(1) + ',' + c[1].toFixed(1)).join(' '), area = path + ` L${coords[coords.length - 1][0].toFixed(1)},${h - pad} L${coords[0][0].toFixed(1)},${h - pad} Z`, dots = coords.map(c => `<circle cx="${c[0].toFixed(1)}" cy="${c[1].toFixed(1)}" r="2.6" fill="var(--accent)"/>`).join('');

      return `<svg viewBox="0 0 ${w} ${h}" width="100%" height="${h}" role="img" aria-label="Evolução do peso"><path d="${area}" fill="var(--accent)" fill-opacity="0.12" stroke="none"/><path d="${path}" fill="none" stroke="var(--accent)" stroke-width="2.4" stroke-linecap="round" stroke-linejoin="round"/>${dots}</svg><div class="chart-range"><span>${min} kg</span><span>${max} kg</span></div>`;

    }

 

    function calNav(delta) { calMonth = new Date(calMonth.getFullYear(), calMonth.getMonth() + delta, 1); render(); }

    function calendarWidget() {

      if (!calSel) calSel = today();

      let y = calMonth.getFullYear(), m = calMonth.getMonth(), first = new Date(y, m, 1), startDow = (first.getDay() + 6) % 7, days = new Date(y, m + 1, 0).getDate(), monthName = calMonth.toLocaleDateString('pt-BR', { month: 'long', year: 'numeric' }), cells = '';

      for (let i = 0; i < startDow; i++) cells += '<div class="cal-cell empty"></div>';

      for (let d = 1; d <= days; d++) {

        let ds = `${y}-${String(m + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}`;

        let dt = DB.tareas.filter(x => x.fecha === ds);

        let mt = DB.metas.filter(x => x.prazo === ds);

        let hasItems = dt.length + mt.length;

        let full = hasItems && dt.every(x => x.hecha) && mt.every(x => x.concluida);

        let isToday = ds === today(), isSel = ds === calSel;

        cells += `<button class="cal-cell ${isToday ? 'today' : ''} ${isSel ? 'sel' : ''}" onclick="calPick('${ds}')" aria-label="${d} de ${monthName}"><span>${d}</span>${hasItems ? `<i class="dot ${full ? 'full' : ''}"></i>` : ''}</button>`;

      }

      let selTasks = DB.tareas.filter(x => x.fecha === calSel);

      let selMetas = DB.metas.filter(x => x.prazo === calSel);

      let agendaHtml = [...selTasks.map(taskItem), ...selMetas.map(metaItem)].join('');

      return `<div class="cal"><div class="cal-head"><button onclick="calNav(-1)" aria-label="Mês anterior">‹</button><b>${monthName}</b><button onclick="calNav(1)" aria-label="Próximo mês">›</button></div><div class="cal-grid">${['S', 'T', 'Q', 'Q', 'S', 'S', 'D'].map(d => `<span class="cal-dow">${d}</span>`).join('')}${cells}</div><div class="cal-agenda">${agendaHtml || `<div class="empty">${calSel === today() ?     }

        if (d.returnValue === 'archive') return act('updateHabito', { id, activo: habitoArchivado }, habitoArchivado ? 'Hábito reativado' : 'Hábito arquivado');

        if (d.returnValue === 'delete') {

          if (!confirm('Excluir este registro? Esta ação não pode ser desfeita.')) return;

          return act('delete' + names[kind], { id }, 'Excluído');

        }

        act((id ? 'update' : 'add') + names[kind], id ? { id, ...p } : p, id ? 'Alterações salvas' : 'Registro adicionado');

      };

    }
